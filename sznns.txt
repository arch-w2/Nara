local function newSquare(props)
    local s = Drawing.new("Square")
    for k, v in pairs(props) do s[k] = v end
    s.Visible = true
    return s
end

local function newText(props)
    local t = Drawing.new("Text")
    for k, v in pairs(props) do t[k] = v end
    t.Visible = true
    return t
end

local _elements = {}
local _refs = {}
local _running = true

local function addElement(obj, dx, dy)
    table.insert(_elements, { obj = obj, dx = dx, dy = dy })
    table.insert(_refs, obj)
    table.insert(_refs, obj)
    return obj
end

local function updateAllPositions()
    for _, e in ipairs(_elements) do
        pcall(function() e.obj.Position = Vector2.new(baseX + e.dx, baseY + e.dy) end)
    end
end

local function destroyAll()
    _running = false
    for _, obj in ipairs(_refs) do
        pcall(function() obj.Visible = false; obj:Destroy() end)
    end
    table.clear(_refs)
    table.clear(_elements)
    print("[MontanaHub] UI cerrada y destruida")
end

local function clampBasePosition()
    local vpW, vpH = getViewportSize()
    baseX = math.clamp(baseX, 0, vpW - CONT_W)
    baseY = math.clamp(baseY, 0, vpH - CONT_H)
end

local UserInputService = game:GetService("UserInputService")
local deviceType, deviceIcon = "Desconocido", "?"

local function detectDevice()
    local hasTouch = UserInputService.TouchEnabled
    local hasMouse = UserInputService.MouseEnabled
    local hasKeyboard = UserInputService.KeyboardEnabled
    local vpW, vpH = getViewportSize()
    
    if hasMouse and hasKeyboard and not hasTouch then
        deviceType, deviceIcon = "PC", "🖥️"
    elseif hasTouch and not hasMouse then
        deviceType, deviceIcon = (vpW > 768 and "Tablet" or "iPhone"), "📱"
    elseif hasTouch and hasMouse then
        deviceType, deviceIcon = "PC Táctil", "💻"
    end
end

local scrollOffset = 0
local maxScroll = math.max(0, (#liveItems - VISIBLE_ROWS) * ROW_H)
local rowBgs, rowTexts, scrollbarThumb = {}, {}, nil

local function clampScroll()
    scrollOffset = math.clamp(scrollOffset, 0, maxScroll)
end

local function updateScroll()
    for i = 1, #liveItems do
        local rowDy = 78 + (i - 1) * ROW_H - scrollOffset
        local isVisible = rowDy >= 78 and rowDy < 78 + SCROLL_AREA_HEIGHT
        
        if rowBgs[i] then
            rowBgs[i].Visible = isVisible
            rowBgs[i].Position = Vector2.new(baseX + 7, baseY + rowDy)
        end
        
        if rowTexts[i] then
            for _, textObj in ipairs(rowTexts[i]) do
                textObj.obj.Visible = isVisible
                textObj.obj.Position = Vector2.new(baseX + textObj.dx, baseY + rowDy + textObj.dyOffset)
            end
        end
    end
    
    if scrollbarThumb and maxScroll > 0 then
        local scrollPercent = scrollOffset / maxScroll
        local thumbY = 75 + scrollPercent * (SCROLL_AREA_HEIGHT - scrollbarThumb.Size.Y)
        scrollbarThumb.Position = Vector2.new(baseX + CONT_W - 12, baseY + thumbY)
    end
end

local mouse = game:GetService("Players").LocalPlayer:GetMouse()
local function isMouseInArea(area) return mouse.X >= area.x1 and mouse.X <= area.x2 and mouse.Y >= area.y1 and mouse.Y <= area.y2 end
local function isPointInArea(px, py, area) return px >= area.x1 and px <= area.x2 and py >= area.y1 and py <= area.y2 end
local function getScrollArea() return { x1 = baseX + 7, y1 = baseY + 75, x2 = baseX + CONT_W - 14, y2 = baseY + 75 + SCROLL_AREA_HEIGHT } end
local function getRowArea(i)
    local rowDy = startDy + (i - 1) * ROW_H - scrollOffset
    return { x1 = baseX + 7, y1 = baseY + rowDy, x2 = baseX + 7 + CONT_W - 20, y2 = baseY + rowDy + ROW_H - 2 }

local dragging, dragStartMouseX, dragStartMouseY, dragStartBaseX, dragStartBaseY = false, 0, 0, 0, 0
local activeTouchId, scrolling, scrollStartY, scrollStartOffset = nil, false, 0, 0

UserInputService.InputChanged:Connect(function(input, gameProcessed)
    if not _running or gameProcessed then return end
    if input.UserInputType == Enum.UserInputType.MouseWheel and isMouseInArea(getScrollArea()) then
        scrollOffset = scrollOffset - input.Position.Z * 30
        clampScroll()
        updateScroll()
    elseif input.UserInputType == Enum.UserInputType.Touch and input.UserInputId == activeTouchId then
        if dragging then
            baseX = dragStartBaseX + (input.Position.X - dragStartMouseX)
            baseY = dragStartBaseY + (input.Position.Y - dragStartMouseY)
            clampBasePosition()
            updateAllPositions()
        elseif scrolling then
            scrollOffset = scrollStartOffset + (scrollStartY - input.Position.Y)
            clampScroll()
            updateScroll()
        end
    end
end)

UserInputService.InputBegan:Connect(function(input, gameProcessed)
    if not _running or gameProcessed then return end
    if input.UserInputType == Enum.UserInputType.MouseButton1 then
        local mx, my = mouse.X, mouse.Y
        if isPointInArea(mx, my, getCloseBtnArea()) then destroyAll(); return end
        if isPointInArea(mx, my, getAserBtnArea()) then
            aserBtnBg.Color = ASER_HOVER_COLOR
            task.delay(0.2, function() if _running then aserBtnBg.Color = Color3.fromRGB(35, 35, 40) end end)
            return
        end
        if isPointInArea(mx, my, getHeaderDragArea()) then
            dragging = true; dragStartMouseX, dragStartMouseY = mx, my; dragStartBaseX, dragStartBaseY = baseX, baseY
        end
    elseif input.UserInputType == Enum.UserInputType.Touch then
        activeTouchId = input.UserInputId
        local tx, ty = input.Position.X, input.Position.Y
        if isPointInArea(tx, ty, getCloseBtnArea()) then destroyAll(); return end
        if isPointInArea(tx, ty, getAserBtnArea()) then
            aserBtnBg.Color = ASER_HOVER_COLOR
            task.delay(0.2, function() if _running then aserBtnBg.Color = Color3.fromRGB(35, 35, 40) end end)
            return
        end
        if isPointInArea(tx, ty, getHeaderDragArea()) then
            dragging = true; dragStartMouseX, dragStartMouseY = tx, ty; dragStartBaseX, dragStartBaseY = baseX, baseY
        elseif isPointInArea(tx, ty, getScrollArea()) then
            scrolling = true; scrollStartY = ty; scrollStartOffset = scrollOffset
        end
    end
end)

UserInputService.InputEnded:Connect(function(input, gameProcessed)
    if input.UserInputType == Enum.UserInputType.MouseButton1 then dragging = false end
    if input.UserInputType == Enum.UserInputType.Touch and input.UserInputId == activeTouchId then
        if dragging then
            local tx, ty = input.Position.X, input.Position.Y
            if math.abs(tx - dragStartMouseX) + math.abs(ty - dragStartMouseY) < 10 then
                for i = 1, #liveItems do
                    if isPointInArea(tx, ty, getRowArea(i)) then
                        if setclipboard then setclipboard(liveItems[i].jobId) end
                        if rowBgs[i] then
                            rowBgs[i].Color = SELECT_COLOR
                            task.delay(0.3, function() if rowBgs[i] and _running then rowBgs[i].Color = BG_COLOR end end)
                        end
                        break
                    end
                end
            end
        end
        dragging, scrolling, activeTouchId = false, false, nil
    end
end)

UserInputService.InputBegan:Connect(function(input, gameProcessed)
    if not _running or gameProcessed or input.UserInputType ~= Enum.UserInputType.MouseButton1 then return end
    if isMouseInArea(getHeaderDragArea()) or isMouseInArea(getCloseBtnArea()) or isMouseInArea(getAserBtnArea()) then return end
    for i = 1, #liveItems do
        if isMouseInArea(getRowArea(i)) then
            if setclipboard then setclipboard(liveItems[i].jobId) end
            if rowBgs[i] then
                rowBgs[i].Color = SELECT_COLOR
                task.delay(0.3, function() if rowBgs[i] and _running then rowBgs[i].Color = BG_COLOR end end)
            end
            break
        end
    end
end)
